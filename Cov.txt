3️⃣ Full CMakeLists.txt with Bullseye Coverage Option

cmake_minimum_required(VERSION 3.16)
project(MyQtApp VERSION 1.0 LANGUAGES CXX)

# Set C++ standard
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# -------------------------
# Bullseye Code Coverage Option
# -------------------------
option(APP_BULLSEYE_CODE_COVERAGE_ENABLED "Enable Bullseye Coverage" ON)

if(APP_BULLSEYE_CODE_COVERAGE_ENABLED)
    message(STATUS "Bullseye Code Coverage is ENABLED")
    
    # Default MinGW path (update if you installed in a different location)
    set(CMAKE_C_COMPILER "C:/Qt/Tools/mingw110_64/bin/gcc.exe")
    set(CMAKE_CXX_COMPILER "C:/Qt/Tools/mingw110_64/bin/g++.exe")
    
    # Bullseye coverage flags (replace with actual Bullseye flags if needed)
    set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS} --coverage")
    set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} --coverage")
endif()

# -------------------------
# Qt6 setup
# -------------------------
set(CMAKE_PREFIX_PATH "C:/Qt/6.5.3/msvc2019_64/lib/cmake") # Update according to your Qt installation

find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# -------------------------
# Add executable
# -------------------------
add_executable(MyQtApp
    main.cpp
    src/mywidget.cpp
)

target_link_libraries(MyQtApp Qt6::Core Qt6::Widgets)

# -------------------------
# Optional: Enable testing
# -------------------------
enable_testing()

4️⃣ Notes on Paths

MinGW default paths (if installed with Qt):

C:/Qt/Tools/mingw110_64/bin/gcc.exe
C:/Qt/Tools/mingw110_64/bin/g++.exe

Qt CMake prefix path:

C:/Qt/6.5.3/msvc2019_64/lib/cmake

Change this according to your actual Qt version and compiler (e.g., MinGW vs MSVC).

Bullseye coverage flags:

--coverage works for GCC/MinGW as a generic example

If using Bullseye Coverage, replace --coverage with Bullseye-specific compile/link flags from your Bullseye license/setup.




---

5️⃣ How to Build

mkdir build
cd build
cmake -G "Ninja" .. -DAPP_BULLSEYE_CODE_COVERAGE_ENABLED=ON
cmake --build .

Check Bullseye .cov files in the build directory after running your program.

Make sure g++.exe is used for C++ compilation; otherwise, coverage may not work.
